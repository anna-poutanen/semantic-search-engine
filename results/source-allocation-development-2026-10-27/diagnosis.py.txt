"""Post hoc descriptive trace analysis only; no candidate execution or imports."""
from collections import Counter
import gzip
import hashlib
import json
from pathlib import Path
import re
import subprocess

ROOT = Path(__file__).resolve().parents[2]
FOLDER = Path(__file__).resolve().parent
FIELDS = ("subset", "dataset", "query_id", "condition")


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def canonical(value):
    return json.dumps(value, sort_keys=True, ensure_ascii=False, allow_nan=False,
                      separators=(",", ":")).encode()


def load(path):
    return json.loads(path.read_bytes())


def identity(row):
    return {field: row[field] for field in FIELDS}


def case_key(row):
    return tuple(row[field] for field in FIELDS)


def physical_lines(source):
    return [line for line in re.findall(r"[^\r\n]*(?:\r\n|\r|\n|$)", source) if line]


def covers(row, target, *, raw=False):
    start, end = target["evidence_lines"]
    source = row["text"] if raw else row["source"]
    return (row["path"] == target["path"] and row["start_line"] <= start <= end <= row["end_line"]
            and sha("".join(physical_lines(source)[start-row["start_line"]:end-row["start_line"]+1]).encode())
                == target["evidence_sha256"])


plan, start = load(FOLDER / "frozen-plan.json"), load(FOLDER / "run-start.json")
summary = load(FOLDER / "summary.json")
manifest = load(FOLDER / "control-references.json")
inputs = {str(path.relative_to(ROOT)): sha(path.read_bytes()) for path in (
    FOLDER / "frozen-plan.json", FOLDER / "run-start.json", FOLDER / "replay.jsonl.gz",
    FOLDER / "control-references.json", FOLDER / "graph-fixtures.json", FOLDER / "summary.json",
    ROOT / manifest["path"], Path(__file__).resolve())}
source_names = ("src/semantic_search/source_allocation.py", "src/semantic_search/backend.py",
                "scripts/replay_source_allocation.py", "scripts/audit_source_allocation.py")
sources = {}
for name in source_names:
    raw = (ROOT / name).read_bytes()
    committed = subprocess.check_output(["git", "-C", str(ROOT), "show", start["execution_commit"] + ":" + name])
    assert sha(raw) == sha(committed) == plan["producer_sha256"][name]
    sources[name] = sha(raw)
assert inputs[str((FOLDER / "frozen-plan.json").relative_to(ROOT))] == start["plan_sha256"]
assert inputs[str((FOLDER / "replay.jsonl.gz").relative_to(ROOT))] == summary["replay_sha256"]

references = {case_key(case): case["controls"] for case in manifest["cases"]}
wanted = {ref["line_number"]: ref for case in references.values() for ref in case.values()}
controls = {}
with gzip.open(ROOT / manifest["path"], "rt", encoding="utf-8") as stream:
    for number, line in enumerate(stream, 1):
        if number not in wanted:
            continue
        row, expected = json.loads(line), wanted[number]
        assert sha(canonical(row)) == expected["record_sha256"]
        assert sha(canonical(row["response"])) == expected["response_sha256"]
        assert sha(row["compact_payload"].encode()) == expected["compact_sha256"]
        controls[(row["selector"], *case_key(row))] = {
            "reference": expected, "compact_payload": row["compact_payload"],
            "response": row["response"], "stages": row["stages"]}
assert len(controls) == 730

counts = {mode: Counter() for mode in ("source_graph", "source_graph_all")}
details, all_guards = [], {}
with gzip.open(FOLDER / "replay.jsonl.gz", "rt", encoding="utf-8") as stream:
    for number, line in enumerate(stream, 1):
        row = json.loads(line)
        mode, key = row["selector"], case_key(row)
        control = controls[("coverage_control", *key)]
        assert row["control_references"] == references[key]
        assert row["reference_response"] == control["response"]
        count = counts[mode]
        count["rows"] += 1
        count["clause_activated"] += row["selection_plan"]["activated"]
        count["planner_accepted"] += row["selection_plan"]["accepted"]
        count["guard_accepted"] += row["source_guard_receipt"]["accepted"]
        count["final_compact_changed_vs_coverage"] += row["compact_payload"] != control["compact_payload"]
        count["complete_gain_rows_vs_coverage"] += row["comparisons"]["coverage_control"]["complete_gain"]
        count["complete_loss_rows_vs_coverage"] += row["comparisons"]["coverage_control"]["complete_loss"]
        count["group_gain_occurrences_vs_coverage"] += len(row["comparisons"]["coverage_control"]["group_gains"])
        count["group_loss_occurrences_vs_coverage"] += len(row["comparisons"]["coverage_control"]["group_losses"])
        if mode == "source_graph_all":
            all_guards[key] = {"ledger_line": number, "record_sha256": sha(canonical(row)),
                               "guard": row["source_guard_receipt"],
                               "final_equals_control": row["compact_payload"] == control["compact_payload"]}
            continue
        changes = row["comparisons"]["coverage_control"]
        if not (changes["group_gains"] or changes["group_losses"]):
            continue
        receipt = row["selection_plan"]
        added_keys = {tuple(receipt["canonical_keys"][identifier]) for identifier in receipt["added_ids"]}
        group_details = []
        for kind in ("group_gains", "group_losses"):
            for change in changes[kind]:
                targets = change["evidence"]
                raw = [{"raw_rank": rank, "id": pair["snippet"]["id"],
                        "path": pair["snippet"]["path"], "symbol": pair["snippet"]["symbol"]}
                       for rank, pair in enumerate(row["stages"]["ranked"], 1)
                       if any(covers(pair["snippet"], target, raw=True) for target in targets)]
                matching_ids = {item["id"] for item in raw}
                baseline_ids = set(receipt["baseline_eligible_ids"])
                eligible_ids = set(receipt["proposed_eligible_ids"])
                classification = ("removed_before_packing" if kind == "group_losses" and matching_ids & set(receipt["evicted_ids"])
                    else "retrieved_but_not_previously_eligible" if kind == "group_gains" and matching_ids and not matching_ids & baseline_ids
                    else "requires_stage_review")
                group_details.append({"change": kind, "group": change["group"], "targets": targets,
                    "raw_covering_windows": raw, "baseline_eligible_covering_ids": sorted(matching_ids & baseline_ids),
                    "proposal_eligible_covering_ids": sorted(matching_ids & eligible_ids), "classification": classification,
                    "baseline_delivered_rank": change["baseline_rank"], "candidate_delivered_rank": change["variant_rank"],
                    "baseline_excerpt_ids": change["baseline_excerpts"], "candidate_excerpt_ids": change["variant_excerpts"]})
        details.append({**identity(row), "ledger_line": number, "record_sha256": sha(canonical(row)),
            "query": row["response"]["query"], "control_references": row["control_references"],
            "baseline_eligible_ids": receipt["baseline_eligible_ids"], "proposed_eligible_ids": receipt["proposed_eligible_ids"],
            "added_ids": receipt["added_ids"], "evicted_ids": receipt["evicted_ids"],
            "added_graph_witnesses": [w for w in receipt["graph_witnesses"] if tuple(w["neighbor_key"]) in added_keys],
            "guard_protected_ids": row["source_guard_receipt"]["protected_ids"],
            "guard_accepted": row["source_guard_receipt"]["accepted"],
            "baseline_evidence_ranks": changes["control_ranks"], "candidate_evidence_ranks": changes["variant_ranks"],
            "complete_gain": changes["complete_gain"], "complete_loss": changes["complete_loss"],
            "baseline_characters": row["reference_response"]["total_source_chars"],
            "baseline_tokens": row["reference_response"]["token_usage"]["tokens"],
            "candidate_characters": row["response"]["total_source_chars"],
            "candidate_tokens": row["response"]["token_usage"]["tokens"],
            "candidate_truncated_ids": [item["id"] for item in row["response"]["results"] if item["truncated"]],
            "backfill": row["proposal_response"]["_selection_backfill"], "group_changes": group_details})

for detail in details:
    detail["all_source_guard_control"] = all_guards[case_key(detail)]
assert counts["source_graph"]["rows"] == counts["source_graph_all"]["rows"] == 365
assert len(details) == 7
assert sum(change["change"] == "group_losses" for detail in details for change in detail["group_changes"]) == 1
assert all(not detail["candidate_truncated_ids"] and detail["backfill"]["trials"] == 0 for detail in details)

graphs = load(FOLDER / "graph-fixtures.json")
httpx_constructor_edges = [entry for fixture in graphs["fixtures"].values() if fixture["dataset"] == "httpx"
                          for entry in fixture["edges"] if entry["caller"] == ["httpx/_models.py", "Request.__init__"]]
assert len(httpx_constructor_edges) == 1
result = {"classification": "POST HOC descriptive analysis of fixed observed outcomes; outside the canonical independent audit scope",
    "execution_commit": start["execution_commit"], "prospective_freeze": start["prospective_freeze"],
    "input_sha256": inputs, "frozen_source_sha256": sources,
    "method": "Read only original JSON traces and frozen graph inventory; verify original record references and source Git blobs. No producer, planner, backend, model, index, retrieval or numerical runtime is imported or executed.",
    "counts": {mode: dict(value) for mode, value in counts.items()},
    "distinct_complete_gains_vs_coverage": summary["qualification"]["source_graph"]["additional_distinct_complete_queries_vs_coverage_control"],
    "changed_group_cases": details, "httpx_constructor_static_edges": httpx_constructor_edges,
    "interpretation": {
        "express": "The app.set caller is already retrieved, then omitted by the old eligible allocation. Explicit calls to compileQueryParser or compileTrust justify its admission. Six complete-case gains share two dataset/query units across repeated studies/modes.",
        "httpx": "Request._prepare is already retrieved at raw rank six and previously delivered completely. Two retained static dependencies occupy slots six/seven, leaving it as the sole exchangeable eighth result. Promoting encode_request (raw rank eight) evicts it before packing. All seven protected rows remain intact; the omitted header-preservation implementation is outside the top-five anchors' one-hop protection.",
        "flow_scope": "Frozen Request.__init__ edges connect both encode_request and Request._prepare. Their constructor-level relationship is beyond the current top-five one-hop rule. This is a post hoc explanation, not an evaluated broader graph rule.",
        "partial_query": "HTTPX lacks the Request.__init__ evidence group before and after. Adding encode_request while dropping Request._prepare swaps one required group for another without changing complete-query counts. Individual group gates correctly reject it.",
        "budgets": "None of the seven changed-group cases clips a result or performs backfill. The HTTPX final response uses 5461/12000 source characters and2223/4096 compact tokens. Its regression is slot eviction rather than a character/token packing error.",
        "all_source_control": "Protecting every actual reference row blocks these exchanges and all measured complete-query gains. Both candidate arms remain unqualified; no production default or fresh run follows."},
    "calls": {"model": 0, "index": 0, "retrieval": 0, "new_candidate_execution": 0, "network": 0},
    "limits": "Derived counts are descriptive and outside canonical audit scope. All365 rows are observed development evidence sharing187 dataset/query units. No fresh source, labels or outcomes are read. No generalization, native latency, coding correctness or provider token claim is made."}
assert all(sha((ROOT / name).read_bytes()) == digest for name, digest in inputs.items())
target = FOLDER / "diagnosis.json"
with target.open("x", encoding="utf-8") as stream:
    json.dump(result, stream, ensure_ascii=False, indent=2, allow_nan=False)
    stream.write("\n")
print(json.dumps({"diagnosis": str(target.relative_to(ROOT)), "sha256": sha(target.read_bytes()), "counts": result["counts"]}))
