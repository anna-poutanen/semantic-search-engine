"""Source-only label authoring after the independent source/scope freeze.

Uses Python's AST and physical source bytes. No model, index, retrieval,
embedding, existing ranking outcome, or HTTPX runtime is opened.
"""
from __future__ import annotations
import ast
import hashlib
import json
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
REPO = ROOT / ".cache/corpora/httpx"
OUT = Path(__file__).parent
sha = lambda raw: hashlib.sha256(raw).hexdigest()
freeze = json.loads((OUT / "source-freeze.json").read_text())
assert not freeze["labels_authored"]


def definitions(path):
    raw = (REPO / path).read_bytes()
    assert sha(raw) == freeze["tracked_python_source_files"][path]
    nodes = {}
    def walk(node, parents):
        for item in ast.iter_child_nodes(node):
            if isinstance(item, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)):
                qualified = ".".join((*parents, item.name))
                if isinstance(item, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    nodes.setdefault(qualified, []).append(item)
                walk(item, (*parents, item.name))
            else:
                walk(item, parents)
    walk(ast.parse(raw.decode("utf-8")), ())
    return raw, nodes


def target(path, symbol):
    raw, nodes = definitions(path)
    assert len(nodes[symbol]) == 1
    node = nodes[symbol][0]
    lines = raw.decode("utf-8").splitlines(keepends=True)
    first, last = node.lineno, node.end_lineno
    evidence = "".join(lines[first - 1:last]).encode()
    return {"path": path, "symbol": symbol, "definition_lines": [first, last],
            "evidence_lines": [first, last],
            "declaration_anchor": lines[first - 1].strip(),
            "evidence_sha256": sha(evidence), "source_file_sha256": sha(raw)}


def present(identifier, query, notes, *items):
    groups = [[target(path, symbol)] for path, symbol in items]
    row = {"id": identifier, "query": query, "kind": "present", "notes": notes,
           "split": "repository_holdout"}
    if len(groups) == 1:
        row["targets"] = groups[0]
    else:
        row["target_groups"] = groups
    return row


def absent(identifier, query, notes):
    return {"id": identifier, "query": query, "kind": "absent", "notes": notes,
            "targets": [], "split": "repository_holdout"}
