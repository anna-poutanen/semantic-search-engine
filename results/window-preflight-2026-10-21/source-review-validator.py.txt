class SourceOnly(importlib.abc.MetaPathFinder):
    def find_spec(self, name, path=None, target=None):
        prefixes = ("mlx", "mlx_embeddings", "torch", "transformers", "jax", "numpy",
                    "semantic_search.backend", "semantic_search.embeddings", "semantic_search.model",
                    "semantic_search.index", "semantic_search.store", "semantic_search.selection",
                    "semantic_search.context_backfill", "semantic_search.window_assembly",
                    "semantic_search.context", "semantic_search.persistent_retrieval")
        if any(name == p or name.startswith(p + ".") for p in prefixes):
            raise RuntimeError("Independent source review forbids " + name)


sys.meta_path.insert(0, SourceOnly())


def deny(*args, **kwargs):
    raise RuntimeError("Independent source review forbids network")

def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def digest(value):
    return sha((json.dumps(value, sort_keys=True, separators=(",", ":")) + "\n").encode())


def git(*args):
    return subprocess.check_output(["git", "-C", str(CORPUS), *args], text=True).strip()

counts = {"queries": len(data["queries"]), "present": 0, "multi_file_present": 0,
          "single_file_present": 0, "scoped_absent": 0, "required_positive_evidence_groups": 0,

targets, tests = [], {}
for query in data["queries"]:
    groups = query["target_groups"]
    if query["kind"] == "absent":
        assert not groups and query["scope"] == data["source_root"]
        counts["scoped_absent"] += 1
        continue
    counts["present"] += 1
    counts["required_positive_evidence_groups"] += len(groups)
    paths = {t["path"] for group in groups for t in group}
    counts["multi_file_present" if len(paths) > 1 else "single_file_present"] += 1
    for group_number, group in enumerate(groups, 1):
        assert len({(t["path"], tuple(t["evidence_lines"]), t["evidence_sha256"]) for t in group}) == 1
        for target in group:
            path = target["path"]
            source = sources[path]
            raw = source.encode()
            lines = source.splitlines(keepends=True)
            low, high = target["evidence_lines"]
            evidence = "".join(lines[low - 1:high]).encode()
            offsets = [0]
            for line in lines:
                offsets.append(offsets[-1] + len(line.encode()))
            assert target["source_file_sha256"] == sha(raw)
            assert target["evidence_sha256"] == sha(evidence)
            assert target["evidence_byte_bounds"] == [offsets[low - 1], offsets[high]]
            matched = [d for d in defs[path] if d.symbol == target["symbol"]
                       and [d.start_line, d.end_line] == target["definition_lines"]
                       and [d.node.start_byte, d.node.end_byte] == target["ast_definition_byte_bounds"]]
            assert len(matched) == 1, (query["id"], target["symbol"])
            definition = matched[0]
            assert [definition.span.start_byte, definition.span.end_byte] == target["definition_span_byte_bounds"]
            assert sha(raw[definition.node.start_byte:definition.node.end_byte]) == target["definition_sha256"]
            assert definition.start_line <= low <= high <= definition.end_line
            assert sorted(set(definition.implementation_lines) & set(range(low, high + 1))) == target["executable_evidence_lines"]
            assert target["executable_evidence_lines"]
            eligible = [w.id for w in windows if w.path == path and w.symbol == target["symbol"]
                        and w.start_line <= low <= high <= w.end_line]
            assert eligible, (query["id"], target["symbol"], low, high)
            targets.append({"query_id": query["id"], "group": group_number,
                            "path": path, "symbol": target["symbol"],
                            "evidence_lines": target["evidence_lines"],
                            "evidence_sha256": sha(evidence),
                            "canonical_definition_key": [path, definition.symbol, definition.start_line,
                                                         definition.end_line, definition.node.start_byte],
                            "eligible_window_ids": eligible})
            counts["target_rows"] += 1
    for reference in query.get("upstream_tests_read", []):
        path, title = reference.split("::", 1)
        if reference in tests:
            continue
        text = (CORPUS / path).read_text()
        raw = text.encode()
        stack, matched = [parse_source(path, text).root_node], []
        while stack:
            node = stack.pop()
            stack.extend(reversed(node.named_children))
            if node.type != "call_expression":
                continue
            fn, args = node.child_by_field_name("function"), node.child_by_field_name("arguments")
            if fn is None or args is None or node_text(fn, raw) not in ("test", "it") or not args.named_children:
                continue
            first = args.named_children[0]
            if first.type == "string" and node_text(first, raw)[1:-1] == title:
                matched.append(node)
        assert len(matched) == 1, reference
        node = matched[0]
        tests[reference] = {"lines": [node.start_point[0] + 1, node.end_point[0] + 1],
                            "call_source_sha256": sha(raw[node.start_byte:node.end_byte]),
                            "full_body_read": True, "executed": False}
assert counts == data["counts"]
