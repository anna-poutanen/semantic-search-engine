ROOT = Path.cwd()
STUDY = ROOT / 'results/qwen-native-mcp-smoke-2026-10-14'
COUNTERS = ('requests', 'attempted_pairs', 'completed_pairs', 'attempted_forwards', 'completed_forwards')
SCORES = ('yes_logit', 'no_logit', 'logit_difference', 'yes_rank_signal')

def require(value, reason):
    if not value:
        raise AssertionError(reason)

def digest(raw):
    return hashlib.sha256(raw).hexdigest()

def file_digest(path):
    out = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            out.update(block)
    return out.hexdigest()

def assignments(path):
    return {node.targets[0].id: node.value for node in ast.parse(path.read_text()).body
            if isinstance(node, ast.Assign) and isinstance(node.targets[0], ast.Name)}

def compact(raw):
    rows = []
    for row in raw['results']:
        rows.append({key: row[key] for key in ('path', 'symbol', 'start_line', 'end_line', 'source', 'source_hash', 'truncated')} |
                    {'origin': row.get('origin', 'retrieval'), 'related_to': row.get('related_to')})
    return {'query': raw['query'], 'match_status': raw['match_status'], 'results': rows,
            'budget_exhausted': raw['budget_exhausted'],
            'token_usage': {key: raw['token_usage'][key] for key in ('encoding', 'tokens', 'budget', 'truncated')},
            'scope': {key: raw['scope'][key] for key in ('source_root', 'membership', 'source_exclusions', 'exclusions_omitted')}}

def scalar(score):
    require(all(type(score[key]) in (int, float) and math.isfinite(score[key]) for key in SCORES), 'finite scalar')
    difference = score['yes_logit'] - score['no_logit']
    signal = 1 / (1 + math.exp(-difference)) if difference >= 0 else math.exp(difference) / (1 + math.exp(difference))
    require(score['logit_difference'] == difference and score['yes_rank_signal'] == signal, 'scalar math')

def validate(result, plan, ledger, source, edited, tokenizer, core, warmup):
    query = plan['fixture']['query']
    defaults, caps = plan['defaults'], plan['caps']
    require(defaults == {'retrieval': 'hybrid', 'rerank': True, 'rerank_provider': 'qwen', 'candidates': 24}, 'defaults')
    require(caps == {'max_results': 8, 'source_budget': 12000, 'token_budget': 4096, 'include_dependencies': True}, 'caps')
    require(result['native_inference'] is True and not result.get('failure') and not result['cleanup_errors'] and not result['interrupt_signals'], 'native run completion')
    require([row['phase'] for row in result['steps']] == ['http_before', 'mcp_before', 'watcher_saved_edit', 'http_after', 'mcp_after', 'mcp_explicit_disable'], 'phase schedule')
    require(result['embedding_counts'] == {'query_calls': 6, 'document_calls': 2, 'document_texts': 62}, 'embedding entry counts')
    require(len(result['observations']) == 6 and result['observations'][0]['response']['indexstats']['indexed_windows'] == 61, 'startup observation')
    schema = result['mcp']['schema']
    require(schema['required'] == ['query'] and schema['additionalProperties'] is False, 'MCP schema query only')
    for name, value in defaults.items() | caps.items():
        require(schema['properties'][name]['default'] == value, 'MCP default schema')
    require(len(ledger) == 194, 'native event inventory')
    pairs = list(zip(ledger[::2], ledger[1::2], strict=True))
    require(all(a['event'] == 'started' and b['event'] == 'completed' for a, b in pairs), 'native event ordering')
    require([b['forward_number'] for _, b in pairs] == list(range(1, 98)), 'native forward numbering')
    require(sum(b['warmup'] is True for _, b in pairs) == 1 and pairs[0][1]['warmup'] is True, 'warmup exclusion')
    for a, b in pairs:
        require(all(b[key] == value for key, value in a.items()), 'native started/completed identities')
        require(b['batch_size'] == 1 and type(b['total_tokens']) is int and 0 < b['total_tokens'] <= 8192, 'unpadded singleton token cap')
        scalar(b['score'])
    rows_by_request = {b['request_id']: b for _, b in pairs}
    require(len(rows_by_request) == 97, 'unique actual native requests')
    encoded_prefix = tokenizer.encode(core['PREFIX'], add_special_tokens=False).ids
    encoded_suffix = tokenizer.encode(core['SUFFIX'], add_special_tokens=False).ids
    checked_native = []
    def input_proof(query, text, receipt, native):
        header = f"<Instruct>: {core['INSTRUCTION']}\n<Query>: {query}\n<Document>: "
        ids = encoded_prefix + tokenizer.encode(header + text, add_special_tokens=False).ids + encoded_suffix
        expected = {'source_sha256': digest(text.encode()), 'delivered_sha256': digest(text.encode()),
                    'query_sha256': digest(query.encode()), 'total_tokens': len(ids)}
        for key, value in expected.items():
            require(receipt[key] == value and native[key] == value, 'literal native token/source identity')
        require(native['input_ids_sha256'] == digest(json.dumps(ids, separators=(',', ':')).encode()), 'tokenizer input IDs')
        require(receipt['truncated'] is False and receipt['source_utf8_bytes'] == receipt['delivered_utf8_bytes'] == len(text.encode()), 'whole source delivery')
        require(receipt['source_lines'] == receipt['delivered_lines'] == len(text.splitlines()), 'physical source lines')
        scalar(receipt)
        require(native['score'] == {key: receipt[key] for key in SCORES}, 'observed native head scalars')
        checked_native.append(native['request_id'])
    input_proof(warmup['query'], warmup['source'], result['owner']['ready']['warmup']['result'], pairs[0][1])
    tokens = {}
    targets = {}
    expected_native = []
    encoder = tiktoken.get_encoding('cl100k_base')
    for index, step in enumerate(result['steps']):
        phase = step['phase']
        if phase == 'watcher_saved_edit':
            stats, watcher = step['status']['indexstats'], step['status']['watcher']
            require(step['trigger_requests'] == 'GET /status only' and step['status_polls'] >= 1, 'saved edit trigger')
            require(stats['indexed_windows'] == 1 and stats['reused_windows'] == 60 and stats['total_windows'] == 61 and stats['deleted_windows'] == 0, 'saved edit incremental index')
            require(watcher['refresh_count'] == 1 and watcher['last_error'] is None, 'watcher refresh')
            require(step['status']['source_manifest'] != result['observations'][0]['response']['indexstats']['source_manifest'], 'source manifest changed')
            continue
        observation = result['observations'][index + 1 if index < 2 else index]
        require(observation['query'] == query, 'actual HTTP query identity')
        response = observation['response']
        current_source = source if phase.endswith('before') else edited
        context = compact(step['context'] if 'context' in step else response)
        require(context['query'] == query and context['match_status'] == 'candidates', 'context query/status')
        count = len(encoder.encode(json.dumps(context, ensure_ascii=False, allow_nan=False, separators=(',', ':')), disallowed_special=()))
        require(count == context['token_usage']['tokens'] and count <= 4096 and context['token_usage']['budget'] == 4096, 'actual compact token cap')
        require(len(context['results']) <= 8 and sum(len(row['source']) for row in context['results']) <= 12000, 'compact source/result cap')
        require(context == compact(response), 'MCP context equals actual HTTP compact response')
        lines = re.findall(r'[^\r\n]*(?:\r\n|\r|\n)|[^\r\n]+$', current_source)
        for row in context['results']:
            require(row['path'] == 'src/sample.py' and row['source'] == ''.join(lines[row['start_line']-1:row['end_line']]), 'saved source span')
            require(row['source_hash'] == digest(row['source'].encode()), 'saved source hash')
        target = [row for row in context['results'] if row['symbol'] == 'backoff_delay']
        require(len(target) == 1 and target[0]['truncated'] is False, 'complete toy target')
        targets[phase] = target[0]['source_hash']
        tokens[phase] = count
        disabled = phase == 'mcp_explicit_disable'
        expected_count = 0 if disabled else 24
        require(all(step['native_after'][key]-step['native_before'][key] == expected_count for key in COUNTERS), 'actual native phase deltas')
        settings = observation['settings']
        for name, value in defaults.items() | caps.items():
            require(settings[name] == (False if disabled and name == 'rerank' else value), 'actual effective HTTP defaults/caps')
        if phase.startswith('mcp'):
            require(step['raw_tool_arguments'] == ({'query': query, 'rerank': False} if disabled else {'query': query}), 'query-only MCP arguments')
            require(step['effective_http_settings'] == settings, 'MCP actual HTTP receipt')
        if disabled:
            require(response['rerank'] is False and not response.get('reranker_receipts'), 'explicit disabled scorer')
            continue
        require(response['candidate_windows'] == 24 and response['retrieval_candidate_windows'] == 50 and response['reranker_batch_size'] == 1 and response['rerank_provider'] == 'qwen', 'actual Qwen prefix/full50')
        receipts = response['reranker_receipts']
        require(len(receipts) == 24, 'actual scored prefix receipt count')
        spans = {}
        for node in ast.parse(current_source).body:
            if isinstance(node, ast.FunctionDef):
                spans[f'src/sample.py:{node.name}:{node.lineno}-{node.end_lineno}'] = ''.join(lines[node.lineno-1:node.end_lineno])
        ordered = sorted(receipts.items(), key=lambda item: item[1]['counts']['completed_forwards'])
        for candidate, receipt in ordered:
            require(candidate in spans and receipt['warmup'] is False and receipt['batch_size'] == 1, 'scored implementation identity')
            number = receipt['counts']['completed_forwards']
            require(all(type(receipt['counts'][key]) is int and receipt['counts'][key] == number for key in COUNTERS), 'actual cumulative native counters')
            native = rows_by_request[receipt['request_id']]
            require(native['forward_number'] == number, 'receipt-native cumulative identity')
            input_proof(query, spans[candidate], receipt['result'], native)
            expected_native.append(receipt['request_id'])
    require(expected_native == [b['request_id'] for _, b in pairs[1:]] and len(checked_native) == 97, 'full actual native request ordering')
    require(targets['http_before'] == targets['mcp_before'] and targets['http_after'] == targets['mcp_after'] and targets['http_before'] != targets['http_after'], 'saved edit source target identity')
    owner = result['owner']
    stop, worker = owner['stop'], owner['worker_stop']
    require(stop['owned_group_stopped'] is True and stop['exit_code'] == 0 and stop['forced_kill'] is False and stop['errors'] == [], 'clean owned process stop')
    require(owner['auth_token_removed'] is True and owner['private_cleanup_errors'] == [] and worker['auth_token_removed'] is True and worker['auth_cleanup_error'] is None and worker['clean_stop'] is True and worker['listener_closed'] is True and result['http_thread_stopped'] is True, 'auth/listener/native cleanup')
    for key in COUNTERS:
        require(type(worker['counts'][key]) is int and worker['counts'][key] == result['native_final_status']['counts'][key] == 97, 'final owned stop counters')
    require(owner['batch_size'] == 1 and owner['score_caching'] is False and owner['query_caching'] is False, 'no cached scoring')
    return {'native_forwards':97, 'warmup_forwards':1, 'measured_forwards':96, 'native_events':194,
            'reconstructed_input_pairs':len(checked_native), 'compact_tokens':tokens, 'target_source_sha256':targets,
            'embedding_counts':result['embedding_counts'], 'saved_edit_indexed_windows':1, 'saved_edit_reused_windows':60}
