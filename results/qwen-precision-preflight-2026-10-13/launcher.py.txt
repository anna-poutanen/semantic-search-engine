"""Root-owned, bounded standalone precision diagnostic launcher."""
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import subprocess
import sys
import time

root = Path.cwd()
sys.path.insert(0, str(root / "scripts"))
from supervise_qwen_optimization import stop_owned

output = root / "results/qwen-head-precision-2026-10-13"
private = root / ".cache/qwen-head-precision-2026-10-13"
private.mkdir(mode=0o700, exist_ok=False)
limit = 1800
started = time.monotonic()
record = {"started_utc": datetime.now(timezone.utc).isoformat(), "timeout_seconds": limit,
          "frozen_plan_sha256": hashlib.sha256((output / "frozen-plan.json").read_bytes()).hexdigest(),
          "launcher_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
          "cleanup_source_sha256": hashlib.sha256((root / "scripts/supervise_qwen_optimization.py").read_bytes()).hexdigest(),
          "owned_sessions": True, "timed_out": False}
model = awake = None
def save(name):
    (output / name).write_text(json.dumps(record, sort_keys=True, indent=2) + "\n")
with (private / "model.log").open("x") as log:
    log_path = private / "model.log"
    log_path.chmod(0o600)
    try:
        model = subprocess.Popen([str(root / ".cache/reranker-experiment/.venv/bin/python"), "-B",
            str(root / "scripts/probe_qwen_batch_precision.py"), "--run",
            "--checkpoint", str(root / ".cache/models/qwen3-reranker-0.6b-original"),
            "--pairs", str(root / "results/qwen-precision-preflight-2026-10-13/pairs.json"),
            "--frozen-plan", str(output / "frozen-plan.json"), "--output", str(output)],
            cwd=root, stdout=log, stderr=subprocess.STDOUT, start_new_session=True)
        record["model_pid"] = model.pid
        awake = subprocess.Popen(["/usr/bin/caffeinate", "-i", "-s", "-w", str(model.pid), "-t", str(limit + 60)],
                                 stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, start_new_session=True)
        record["caffeinate_pid"] = awake.pid
        save("supervisor-start.json")
        print(json.dumps({"model_pid": model.pid, "precision_probe_started": True}), flush=True)
        model.wait(timeout=limit)
        if model.returncode != 0:
            raise RuntimeError("Precision diagnostic exited unsuccessfully")
    except BaseException as error:
        record["error_type"], record["error"] = type(error).__name__, str(error)
        record["timed_out"] = isinstance(error, subprocess.TimeoutExpired)
    finally:
        if model is not None:
            record["model_stop"] = stop_owned(model)
        if awake is not None:
            record["caffeinate_stop"] = stop_owned(awake)
        record["elapsed_seconds"] = time.monotonic() - started
        record["finished_utc"] = datetime.now(timezone.utc).isoformat()
        record["success"] = ("error" not in record and record.get("model_stop", {}).get("exit_code") == 0
            and record.get("model_stop", {}).get("owned_group_stopped") is True
            and record.get("caffeinate_stop", {}).get("owned_group_stopped") is True
            and not record.get("model_stop", {}).get("forced_kill")
            and not record.get("caffeinate_stop", {}).get("forced_kill"))
        save("supervisor.json")
        print(json.dumps(record), flush=True)
sys.exit(0 if record["success"] else 1)
